01 — เกี่ยวกับโครงงาน

เซนเซอร์ตัวเล็ก จอในตัว ครบในบอร์ดเดียว

โครงงานนี้ใช้อุปกรณ์เพียงสองชิ้นหลัก คือเซนเซอร์วัดสภาพอากาศราคาประหยัด และบอร์ดพัฒนาที่มีจอแสดงผลติดมาให้แล้ว จึงเหมาะสำหรับผู้เริ่มต้นที่อยากทำสถานีวัดอุณหภูมิของตัวเอง

DHT11 — เซนเซอร์วัดอุณหภูมิ/ความชื้น

เซนเซอร์ดิจิทัลราคาถูก วัดได้ทั้งอุณหภูมิและความชื้นในตัวเดียว ส่งค่าออกมาเป็นสัญญาณดิจิทัลผ่านขาเดียว จึงต่อสายง่าย เหมาะกับงานที่ไม่ต้องการความละเอียดสูงมาก เช่น สถานีอากาศในบ้าน หรือระบบแจ้งเตือนอุณหภูมิห้อง

ESP32 CYD — บอร์ดพร้อมจอในตัว

ชื่อเล่นของบอร์ด ESP32-2432S028R ซึ่งมีจอสัมผัส TFT ขนาด 2.8 นิ้วติดตั้งมาพร้อมแผงวงจร ทำให้แสดงผลค่าที่อ่านได้จากเซนเซอร์ได้ทันทีโดยไม่ต้องต่อจอเพิ่ม และยังเชื่อมต่อ Wi-Fi เพื่อส่งข้อมูลออนไลน์ได้ในอนาคต

สเปกของอุปกรณ์หลัก

ข้อมูลจำเพาะที่ใช้อ้างอิงตอนต่อวงจรและเขียนโค้ด

03 — วงจรการเชื่อมต่อ

ต่อสาย 3 เส้น ก็เริ่มอ่านค่าได้

DHT11 มีเพียง 3 ขา จึงต่อเข้ากับหัวต่อ GPIO ของ ESP32 CYD ได้โดยตรง ไม่ต้องใช้เบรดบอร์ดเพิ่ม

VCC → 3V3 DATA → GPIO22 GND → GND

04 — โค้ดตัวอย่าง

อ่านค่าเซนเซอร์แล้วแสดงบนจอ

ใช้ไลบรารี DHT sensor library อ่านค่า และ TFT_eSPI วาดผลลัพธ์บนจอของบอร์ด

sensor_display.ino
// อ่านค่าจาก DHT11 แล้วแสดงบนจอ ESP32 CYD #include <DHT.h> #include <TFT_eSPI.h> #define DHTPIN 22 #define DHTTYPE DHT11 DHT dht(DHTPIN, DHTTYPE); TFT_eSPI tft = TFT_eSPI(); void setup() { dht.begin(); tft.init(); tft.setRotation(1); tft.fillScreen(TFT_BLACK); } void loop() { float temp = dht.readTemperature(); float hum = dht.readHumidity(); if (isnan(temp) || isnan(hum)) { tft.drawString("อ่านค่าเซนเซอร์ไม่สำเร็จ", 20, 20, 2); return; } tft.fillScreen(TFT_BLACK); tft.setTextColor(TFT_ORANGE); tft.drawString(String(temp, 1) + " C", 20, 40, 6); tft.setTextColor(TFT_CYAN); tft.drawString("ความชื้น " + String(hum, 0) + "%", 20, 110, 4); delay(2000); }
