01 — เกี่ยวกับโครงงาน

เซนเซอร์ตัวเล็ก จอในตัว ครบในบอร์ดเดียว

โครงงานนี้ใช้อุปกรณ์เพียงสองชิ้นหลัก คือเซนเซอร์วัดสภาพอากาศราคาประหยัด และบอร์ดพัฒนาที่มีจอแสดงผลติดมาให้แล้ว จึงเหมาะสำหรับผู้เริ่มต้นที่อยากทำสถานีวัดอุณหภูมิของตัวเอง

DHT11 — เซนเซอร์วัดอุณหภูมิ/ความชื้น

เซนเซอร์ดิจิทัลราคาถูก วัดได้ทั้งอุณหภูมิและความชื้นในตัวเดียว ส่งค่าออกมาเป็นสัญญาณดิจิทัลผ่านขาเดียว จึงต่อสายง่าย เหมาะกับงานที่ไม่ต้องการความละเอียดสูงมาก เช่น สถานีอากาศในบ้าน หรือระบบแจ้งเตือนอุณหภูมิห้อง

ESP32 CYD — บอร์ดพร้อมจอในตัว

ชื่อเล่นของบอร์ด ESP32-2432S028R ซึ่งมีจอสัมผัส TFT ขนาด 2.8 นิ้วติดตั้งมาพร้อมแผงวงจร ทำให้แสดงผลค่าที่อ่านได้จากเซนเซอร์ได้ทันทีโดยไม่ต้องต่อจอเพิ่ม และยังเชื่อมต่อ Wi-Fi เพื่อส่งข้อมูลออนไลน์ได้ในอนาคต

02 — อุปกรณ์ที่ใช้

สเปกของอุปกรณ์หลัก

ข้อมูลจำเพาะที่ใช้อ้างอิงตอนต่อวงจรและเขียนโค้ด

DHT11 · เซนเซอร์วัดอุณหภูมิ/ความชื้น
ช่วงอุณหภูมิ0–50°C (±2°C)
ช่วงความชื้น20–90% RH (±5%)
แรงดันไฟเลี้ยง3.3–5.5 V
รูปแบบสัญญาณDigital, single-wire
อัตราการอ่านค่า~1 ครั้ง/วินาที
ESP32-2432S028R (ESP32 CYD)
ชิปหลักESP32-WROOM-32
จอแสดงผลTFT 2.8" ILI9341, 320×240
ระบบสัมผัสResistive, XPT2046
การเชื่อมต่อWi-Fi, Bluetooth
ไฟเข้าบอร์ด5V ผ่าน Micro-USB

03 — วงจรการเชื่อมต่อ

ต่อสาย 3 เส้น ก็เริ่มอ่านค่าได้

DHT11 มีเพียง 3 ขา จึงต่อเข้ากับหัวต่อ GPIO ของ ESP32 CYD ได้โดยตรง ไม่ต้องใช้เบรดบอร์ดเพิ่ม

VCC → 3V3 DATA → GPIO22 GND → GND

04 — โค้ดตัวอย่าง

อ่านค่าเซนเซอร์แล้วแสดงบนจอ

ใช้ไลบรารี DHT sensor library อ่านอุณหภูมิ และ LovyanGFX วาดผลลัพธ์บนจอ ILI9341 ของบอร์ด ESP32 CYD โดยตั้งค่าขา SPI ของจอและขา DATA ของ DHT11 ไว้ที่ GPIO22

esp32_cyd_dht11.ino
#include <Arduino.h> #include <LovyanGFX.hpp> #include <DHT.h> #include <math.h> // ===================================================== // ตั้งค่า DHT11 // ===================================================== static constexpr uint8_t DHT_PIN = 22; #define DHT_TYPE DHT11 DHT dht(DHT_PIN, DHT_TYPE); // DHT11 ไม่ควรอ่านถี่กว่า 2 วินาที static constexpr unsigned long SENSOR_INTERVAL = 2500; unsigned long previousSensorTime = 0; // ===================================================== // ตั้งค่าจอ ESP32-2432S028R CYD // จอ ILI9341 ขนาด 2.8 นิ้ว // ===================================================== class LGFX : public lgfx::LGFX_Device { private: lgfx::Panel_ILI9341 panel; lgfx::Bus_SPI bus; lgfx::Light_PWM backlight; public: LGFX() { // ตั้งค่า SPI ของจอ { auto config = bus.config(); config.spi_host = VSPI_HOST; config.spi_mode = 0; config.freq_write = 40000000; config.freq_read = 16000000; config.spi_3wire = false; config.use_lock = true; config.dma_channel = 1; config.pin_sclk = 14; config.pin_mosi = 13; config.pin_miso = 12; config.pin_dc = 2; bus.config(config); panel.setBus(&bus); } // ตั้งค่าจอ ILI9341 { auto config = panel.config(); config.pin_cs = 15; config.pin_rst = -1; config.pin_busy = -1; config.memory_width = 240; config.memory_height = 320; config.panel_width = 240; config.panel_height = 320; config.offset_x = 0; config.offset_y = 0; config.offset_rotation = 0; config.readable = true; config.invert = false; config.rgb_order = false; config.dlen_16bit = false; config.bus_shared = true; panel.config(config); } // ตั้งค่าไฟหน้าจอ { auto config = backlight.config(); config.pin_bl = 21; config.invert = false; config.freq = 44100; config.pwm_channel = 7; backlight.config(config); panel.setLight(&backlight); } setPanel(&panel); } }; LGFX display; // ===================================================== // เลือกสีตามอุณหภูมิ // ===================================================== uint16_t getTemperatureColor(float temperature) { if (temperature < 20.0f) { return TFT_CYAN; } if (temperature < 30.0f) { return TFT_GREEN; } if (temperature < 40.0f) { return TFT_YELLOW; } return TFT_RED; } // ===================================================== // วาดโครงหน้าจอ // ===================================================== void drawScreenLayout() { display.fillScreen(TFT_BLACK); // แถบหัวข้อด้านบน display.fillRect(0, 0, 320, 42, TFT_DARKCYAN); display.setTextDatum(textdatum_t::middle_center); display.setTextColor(TFT_WHITE, TFT_DARKCYAN); display.setTextSize(2); display.drawString( "TEMPERATURE", 160, 21 ); // กรอบแสดงอุณหภูมิ display.drawRoundRect( 10, 55, 300, 135, 12, TFT_RED ); // ป้าย TEMPERATURE display.setTextDatum(textdatum_t::top_left); display.setTextSize(1); display.setTextColor(TFT_LIGHTGREY, TFT_BLACK); display.drawString( "DHT11 TEMPERATURE", 22, 66 ); // ข้อความด้านล่าง display.setTextDatum(textdatum_t::middle_center); display.setTextSize(1); display.setTextColor(TFT_ORANGE, TFT_BLACK); display.drawString( "DHT11 DATA: GPIO22", 160, 218 ); } // ===================================================== // ล้างพื้นที่แสดงอุณหภูมิ // ===================================================== void clearTemperatureArea() { display.fillRect( 17, 88, 286, 85, TFT_BLACK ); } // ===================================================== // แสดงข้อความระหว่างเริ่มต้น // ===================================================== void showStartingMessage() { clearTemperatureArea(); display.setTextDatum(textdatum_t::middle_center); display.setTextSize(2); display.setTextColor(TFT_YELLOW, TFT_BLACK); display.drawString( "Starting...", 160, 115 ); display.setTextSize(1); display.setTextColor(TFT_LIGHTGREY, TFT_BLACK); display.drawString( "Waiting for DHT11", 160, 148 ); } // ===================================================== // แสดงค่าอุณหภูมิบนหน้าจอ // ===================================================== void updateDisplay(float temperature) { clearTemperatureArea(); uint16_t temperatureColor = getTemperatureColor(temperature); display.setTextDatum(textdatum_t::middle_center); display.setTextSize(4); display.setTextColor( temperatureColor, TFT_BLACK ); display.drawString( String(temperature, 1) + " C", 160, 128 ); } // ===================================================== // แสดงข้อผิดพลาดของเซ็นเซอร์ // ===================================================== void showSensorError() { clearTemperatureArea(); display.setTextDatum(textdatum_t::middle_center); display.setTextSize(2); display.setTextColor(TFT_RED, TFT_BLACK); display.drawString( "READ ERROR", 160, 112 ); display.setTextSize(1); display.setTextColor(TFT_YELLOW, TFT_BLACK); display.drawString( "Check DHT11 wiring", 160, 142 ); display.setTextColor(TFT_LIGHTGREY, TFT_BLACK); display.drawString( "DATA = GPIO22", 160, 160 ); } // ===================================================== // แสดงอุณหภูมิใน Serial Monitor // ===================================================== void printSerialData(float temperature) { Serial.print("Temperature: "); Serial.print(temperature, 1); Serial.println(" C"); } // ===================================================== // Setup // ===================================================== void setup() { Serial.begin(115200); delay(500); // เริ่มต้นหน้าจอ display.init(); display.setRotation(1); display.setBrightness(255); drawScreenLayout(); showStartingMessage(); // เริ่มต้น DHT11 dht.begin(); Serial.println(); Serial.println("ESP32 CYD + DHT11 Temperature Monitor"); Serial.println("DHT11 DATA pin: GPIO22"); Serial.println("Waiting for sensor..."); // รอเซ็นเซอร์เริ่มทำงาน delay(2000); // ทำให้อ่านค่าทันทีเมื่อเข้า loop previousSensorTime = millis() - SENSOR_INTERVAL; } // ===================================================== // Loop // ===================================================== void loop() { unsigned long currentTime = millis(); if (currentTime - previousSensorTime >= SENSOR_INTERVAL) { previousSensorTime = currentTime; // อ่านเฉพาะอุณหภูมิ float temperature = dht.readTemperature(); // ตรวจสอบว่าอ่านค่าสำเร็จหรือไม่ if (isnan(temperature)) { Serial.println("อ่านอุณหภูมิ DHT11 ไม่สำเร็จ"); showSensorError(); return; } printSerialData(temperature); updateDisplay(temperature); } }
